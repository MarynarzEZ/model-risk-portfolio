# Pinned to exact versions so the project stays reproducible.
# pandas 3.x and numpy 2.x changed several APIs, so an unpinned install
# could break this code months from now. Alternative would be a loose
# lower bound (e.g. pandas>=2.0), which is friendlier to other projects
# but gives no guarantee that the numbers below reproduce exactly.

# Numerical core
numpy==2.5.3
pandas==3.0.5

# Statistics: distributions and tests (Student's t, normality, chi-square)
scipy==1.18.1

# Market data
yfinance==1.7.0

# Plotting
matplotlib==3.11.2

# Testing. Only needed to run the test suite, not to use the model, but
# kept in one file rather than split into requirements-dev.txt: this is
# a single-developer project and two files to keep in sync is the more
# likely source of trouble.
pytest==9.0.1

# The GUI. Streamlit pulls in pyarrow, altair and uvicorn; they are
# transitive and deliberately not pinned here, so that a Streamlit
# upgrade is not blocked by a stale pin on something it owns.
streamlit==1.67.0
